### 共通CTE（4章・4-2・3-1・保有顧客数）
```sql
WITH paid AS (
  SELECT so.customer_id, so.name AS order_name, DATE(so.created_at) AS d, so.discount_codes, so.line_items
  FROM `spic-core.public.shopify_orders` so
  WHERE so.test=0 AND so.cancelled_at IS NULL AND so.customer_id IS NOT NULL
    AND (so.total_price - so.current_total_tax) > 0
    AND (so.discount_codes IS NULL OR so.discount_codes NOT LIKE '%CYS0810%')
    AND DATE(so.created_at) <= '2026-09-30'),
line2 AS (
  SELECT p.customer_id, p.order_name, p.d, JSON_EXTRACT_SCALAR(item,'$.sku') AS sku,
    (JSON_EXTRACT_SCALAR(item,'$.sku')='1004496' AND p.discount_codes LIKE '%OGS2025_11%') AS is_ogs,
    IF(JSON_EXTRACT_SCALAR(item,'$.sku')='1004496' AND p.discount_codes LIKE '%OGS2025_11%', 0,
       SAFE_CAST(JSON_EXTRACT_SCALAR(item,'$.price') AS FLOAT64)) AS pr,
    SAFE_CAST(JSON_EXTRACT_SCALAR(item,'$.quantity') AS FLOAT64) AS qty
  FROM paid p, UNNEST(JSON_EXTRACT_ARRAY(p.line_items)) item),
life AS (SELECT customer_id, MIN(d) AS life_first FROM paid GROUP BY 1),
bo AS (
  SELECT l.customer_id, l.order_name, l.d, MAX(IF(bm.product_type='定期',1,0)) AS is_teiki
  FROM line2 l JOIN `spic-com-2025-apr-00.sandbox_nishimori.shopify_sku_brand_master` bm ON l.sku = CAST(bm.sku AS STRING)
  WHERE bm.brand IN ('VC','C+D','C+D+K') AND l.pr > 0
  GROUP BY 1,2,3),
ranked AS (SELECT *, ROW_NUMBER() OVER (PARTITION BY customer_id ORDER BY d, order_name) AS rn FROM bo),
cust AS (
  SELECT r.customer_id, r.d AS first_date, r.is_teiki, lf.life_first
  FROM ranked r JOIN life lf USING (customer_id) WHERE r.rn = 1),
trial AS (
  SELECT customer_id, MIN(d) AS origin, LOGICAL_OR(is_ogs) AS has_ogs
  FROM line2 WHERE sku IN ('3023-cp','3031-cp') OR is_ogs GROUP BY 1)
```
### 4章（q4）：`common.sql`の末尾に連結

SELECT 'cohort' AS kind, FORMAT_DATE('%Y-%m', DATE_TRUNC(first_date, MONTH)) AS ym,
  COUNT(*) AS c1_m0, COUNTIF(first_date=life_first) AS c2_pure, COUNTIF(first_date>life_first) AS c3_item,
  COUNTIF(first_date=life_first AND is_teiki=1) AS c4_pure_sel, COUNTIF(first_date>life_first AND is_teiki=1) AS c5_item_sel
FROM cust WHERE first_date >= '2024-05-01' GROUP BY 2
UNION ALL
SELECT 'trial', FORMAT_DATE('%Y-%m', DATE_TRUNC(t.origin, MONTH)),
  COUNT(*), COUNTIF(c.first_date IS NULL), COUNTIF(c.first_date IS NOT NULL), COUNTIF(t.has_ogs), COUNTIF(c.first_date IS NOT NULL AND c.first_date = t.origin)
FROM trial t LEFT JOIN cust c ON c.customer_id = t.customer_id
WHERE c.first_date IS NULL OR c.first_date >= t.origin
GROUP BY 2
ORDER BY 1, 2

### 7-2章（グロスフロー・顧客単位×和集合）
```sql
WITH v AS (SELECT CAST(customer_id AS STRING) AS cid, snapshot_ym AS ym, UPPER(status) AS st, (has_vc=1 OR has_cd=1 OR has_cdk=1) AS u
  FROM `spic-com-2025-apr-00.sandbox_nishimori.subscription_history_view` WHERE snapshot_ym >= '2025-05'),
cm AS (SELECT cid, ym,
    CASE WHEN LOGICAL_OR(u AND st='ACTIVE') THEN 'A' WHEN LOGICAL_OR(u AND st='PAUSED') THEN 'P'
         WHEN LOGICAL_OR(u AND st='CANCELLED') THEN 'C' ELSE 'N' END AS s,
    LOGICAL_OR((NOT u) AND st='ACTIVE') AS other_active
  FROM v GROUP BY 1,2),
months AS (SELECT DISTINCT ym, FORMAT_DATE('%Y-%m', DATE_SUB(PARSE_DATE('%Y-%m-%d', CONCAT(ym,'-01')), INTERVAL 1 MONTH)) AS pym FROM cm),
cur AS (SELECT cm.cid, cm.ym, cm.s, cm.other_active, m.pym FROM cm JOIN months m USING (ym)),
j AS (SELECT COALESCE(c.ym, m2.ym) AS ym, COALESCE(c.cid, p.cid) AS cid, IFNULL(p.s,'X') AS ps, IFNULL(c.s,'X') AS cs, IFNULL(c.other_active, FALSE) AS c_other_active
  FROM cur c FULL OUTER JOIN (SELECT cm.cid, cm.s, m.ym FROM cm JOIN months m ON cm.ym = m.pym) p ON c.cid = p.cid AND c.ym = p.ym
  LEFT JOIN (SELECT DISTINCT ym FROM months) m2 ON m2.ym = p.ym)
SELECT ym, COUNTIF(cs='A') AS stock, COUNTIF(cs='A' AND ps!='A') AS new_active, COUNTIF(cs='A' AND ps='X') AS na_first_seen,
  COUNTIF(cs='A' AND ps='P') AS na_ret_pause, COUNTIF(cs='A' AND ps='C') AS na_ret_cancel, COUNTIF(cs='A' AND ps='N') AS na_from_other_brand,
  COUNTIF(ps='A' AND cs!='A') AS lost_active, COUNTIF(ps='A' AND cs='P') AS to_pause, COUNTIF(ps='A' AND cs='C') AS to_cancel,
  COUNTIF(ps='A' AND cs='N' AND c_other_active) AS to_other_brand_active, COUNTIF(ps='A' AND cs='N' AND NOT c_other_active) AS to_other_brand_inactive,
  COUNTIF(ps='A' AND cs='X') AS to_vanished
FROM j WHERE ym >= '2025-06' GROUP BY ym ORDER BY ym
```
